## 三、树莓派连接步骤

### 3.1 硬件准备

| 配件 | 要求 |
|------|------|
| 树莓派 | 4B 或 Jetson Nano |
| 摄像头 | 树莓派官方摄像头 |
| 网络 | WiFi 或有线网络 |
| 电源 | 5V 3A |
| SD 卡 | 16GB 及以上 |

### 3.2 系统烧录

1. 下载 Raspberry Pi OS
   - 官网：https://www.raspberrypi.com/software/
   - 使用 Raspberry Pi Imager 烧录

2. 配置 WiFi（headless 模式）
   - 在 boot 分区创建 `wpa_supplicant.conf`
   ```
   country=CN
   ctrl_interface=DIR=/var/run/wpa_supplicant GROUP=netdev
   update_config=1
   network={
       ssid="你的 WiFi 名称"
       psk="你的 WiFi 密码"
   }
   ```

3. 启用 SSH
   - 在 boot 分区创建空文件 `ssh`

### 3.3 树莓派配置

```bash
# SSH 登录树莓派
ssh pi@raspberrypi.local

# 更新系统
sudo apt update && sudo apt upgrade -y

# 启用摄像头
sudo raspi-config
# 选择 Interface Options -> Camera -> Enable

# 重启
sudo reboot
```

### 3.5 设备注册到服务器

```python
#!/usr/bin/env python3
# save as: /home/pi/smart_doorbell/device_register.py

import requests
import json
import uuid

# 获取设备唯一 ID
def get_device_id():
    # 使用树莓派序列号
    with open('/proc/cpuinfo', 'r') as f:
        for line in f:
            if line.startswith('Serial'):
                return line.split(':')[1].strip()
    return str(uuid.getnode())

# 注册设备到服务器
def register_device():
    device_id = get_device_id()
    device_name = "客厅门铃"
    
    # 服务器 API
    server_url = "http://8.134.196.56:5000"
    
    # 需要先登录获取 token
    login_data = {
        "username": "admin",
        "password": "Admin@123456"
    }
    resp = requests.post(f"{server_url}/api/auth/login", json=login_data)
    token = resp.json()['data']['access_token']
    
    # 注册设备
    register_data = {
        "device_id": device_id,
        "device_name": device_name,
        "device_type": "raspberry_pi_4b"
    }
    headers = {"Authorization": f"Bearer {token}"}
    resp = requests.post(
        f"{server_url}/api/device/register",
        json=register_data,
        headers=headers
    )
    print(f"设备注册结果：{resp.json()}")

if __name__ == "__main__":
    register_device()
```

### 3.6 设备心跳程序

```python
#!/usr/bin/env python3
# save as: /home/pi/smart_doorbell/heartbeat.py

import requests
import time
import json

SERVER_URL = "http://8.134.196.56:5000"
DEVICE_ID = "你的设备 ID"  # 从注册时获取
TOKEN = "你的访问令牌"

def send_heartbeat():
    """每 30 秒发送一次心跳"""
    while True:
        try:
            data = {
                "device_id": DEVICE_ID,
                "status": "online",
                "battery_level": 100,  # 如果是电池供电
                "wifi_strength": -50   # WiFi 信号强度
            }
            headers = {"Authorization": f"Bearer {TOKEN}"}
            resp = requests.post(
                f"{SERVER_URL}/api/device/heartbeat",
                json=data,
                headers=headers
            )
            print(f"心跳成功：{resp.json()}")
        except Exception as e:
            print(f"心跳失败：{e}")
        
        time.sleep(30)

if __name__ == "__main__":
    send_heartbeat()
```

### 3.7 人脸采集程序

```python
#!/usr/bin/env python3
# save as: /home/pi/smart_doorbell/face_capture.py

import cv2
import requests
import time

SERVER_URL = "http://8.134.196.56:5000"
DEVICE_ID = "你的设备 ID"

def capture_face():
    """采集人脸并上传到服务器"""
    cap = cv2.VideoCapture(0)
    
    # 等待摄像头预热
    time.sleep(1)
    
    # 捕获一帧
    ret, frame = cap.read()
    if ret:
        # 保存为临时文件
        temp_path = "/tmp/face_capture.jpg"
        cv2.imwrite(temp_path, frame)
        
        # 上传到服务器
        with open(temp_path, 'rb') as f:
            files = {'photo': f}
            data = {
                'device_id': DEVICE_ID,
                'visitor_type': 'family'  # 或 'stranger'
            }
            resp = requests.post(
                f"{SERVER_URL}/api/visitor/upload",
                files=files,
                data=data
            )
            print(f"上传结果：{resp.json()}")
    
    cap.release()

if __name__ == "__main__":
    capture_face()
```
